// Swipe right: join the session (Supabase table protected by RLS)
async function commitSwipeDecision(sess, liked){
  if(liked){
    if(!myProfile){ openWizard(); return; }
    const { error } = await sb.from('session_joiners')
      .insert({ session_id: sess.id, user_id: myProfile.id });
    if(!error){
      sess.joiners = sess.joiners || [];
      sess.joiners.push(myProfile.name);
      createNotif(sess, myProfile.name); // notifies the session's author
    }
  } else {
    // Swipe left: the session won't be shown again
    passedSessionIds.push(sess.id);
    await savePassed();
  }
  renderFeed();
}
// Calorie needs: Mifflin-St Jeor formula + activity level
function computeBMR(poids, taille, age, sexe){
  if(!poids || !taille || !age) return null;
  const base = 10 * poids + 6.25 * taille - 5 * age;
  return Math.round(sexe === 'Femme' ? base - 161 : base + 5);
}

function computeTDEE(bmr, activityLevel){
  return Math.round(bmr * (ACTIVITY_MULTIPLIERS[activityLevel] || 1.375));
}
